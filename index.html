Conectar ao Google Sheets

1. No Apps Script da planilha, adicione esta função:

function doGet() { const ss = SpreadsheetApp.getActiveSpreadsheet(); const abaRank = ss.getSheetByName("Ranking"); const abaPont = ss.getSheetByName("Pontuação"); const hoje = new Date(); const diaHoje = hoje.getDate(); const mesHoje = hoje.getMonth(); const anoHoje = hoje.getFullYear(); const ptsDia = {}; if (abaPont) { const rows = abaPont.getDataRange().getValues(); rows.slice(1).forEach(r => { const carimbo = new Date(r[2]); if (carimbo.getDate() === diaHoje && carimbo.getMonth() === mesHoje && carimbo.getFullYear() === anoHoje) { const id = String(r[3] || "").trim(); const total = Number(r[9] || 0); if (id) ptsDia[id] = (ptsDia[id] || 0) + total; } }); } const dados = abaRank.getDataRange().getValues(); const resultado = dados.slice(1).map(r => ({ pos: r[0], id: r[1], nome: r[2], pts: r[3], pts_hoje: ptsDia[String(r[1] || "").trim()] || 0 })); return ContentService .createTextOutput(JSON.stringify(resultado)) .setMimeType(ContentService.MimeType.JSON); }

2. Clique em Implantar → Nova implantação → App da Web

3. Em "Quem tem acesso" selecione "Qualquer pessoa" e copie a URL

4. Cole a URL no campo da página e clique em Carregar
